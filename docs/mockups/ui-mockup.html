SELECT   c.id,
        c.full_name,
        c.email,
        a.balance,
        a.status AS account_status,
        k.kyc_status
FROM on_boarding.customers c
JOIN account_management.accounts a   ON c.id = a.user_id
JOIN on_boarding.kyc_requests k     ON c.id = k.customer_id
WHERE a.status = 'active'
  AND k.kyc_status = 'approved'
ORDER BY a.balance DESC
LIMIT 100
